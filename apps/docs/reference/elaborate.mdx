---
title: "elaborate()"
description: "Flatten a hierarchical definition into primitive gates and the wires between them."
icon: "network"
---

```ts
function elaborate(definition: ComponentDefinition): Netlist;
```

Returns the flattened, deeply frozen and serializable graph that the simulator runs. Results are cached per definition. Throws `CircuitError` if the definition can't be elaborated, for example because of a wire loop with no gate in it.

## Netlist

<ResponseField name="root" type="string">
  Always `"root"`.
</ResponseField>

<ResponseField name="definition" type="string">
  The name of the elaborated definition.
</ResponseField>

<ResponseField name="inputs" type="Record<string, string>">
  External input name to wire id.
</ResponseField>

<ResponseField name="outputs" type="Record<string, string>">
  External output name to wire id.
</ResponseField>

<ResponseField name="wires" type="WireInfo[]">
  Every wire, sorted by id. A wire is one driver plus every port connected to it.

  <Expandable title="WireInfo">
    <ResponseField name="id" type="string">The wire id: the path of its driving port.</ResponseField>
    <ResponseField name="driver" type="'input' | 'gate'">Whether a root input or a primitive output drives the wire.</ResponseField>
    <ResponseField name="gate" type="string | null">The path of the driving gate, or `null` for inputs.</ResponseField>
    <ResponseField name="ports" type="string[]">Every port path carrying this wire's value, in hierarchical order.</ResponseField>
    <ResponseField name="readers" type="string[]">The paths of the gates that read this wire.</ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="gates" type="GateInfo[]">
  Every primitive gate, sorted by path.

  <Expandable title="GateInfo">
    <ResponseField name="path" type="string">Hierarchical path, for example `root.not.g`.</ResponseField>
    <ResponseField name="definition" type="string">The primitive's definition name.</ResponseField>
    <ResponseField name="ancestors" type="{ id: string; definition: string }[]">The composites that contain the gate, outermost first.</ResponseField>
    <ResponseField name="inputs" type="Record<string, string>">Input port name to wire id.</ResponseField>
    <ResponseField name="outputs" type="Record<string, string>">Output port name to wire id.</ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="ports" type="string[]">
  Every port path in the hierarchy, in hierarchical (declaration) order. This is the same key order as `inspect().signals`.
</ResponseField>

<ResponseField name="portWires" type="Record<string, string>">
  Port path to wire id.
</ResponseField>

## Example

```ts
import { elaborate } from "@pbzona/wires";
import { AND } from "@pbzona/wires/components";

const netlist = elaborate(AND);
const wire = netlist.wires.find((w) => w.id === "root.nand.outputs.out");
console.log(wire?.ports);
// ["root.nand.outputs.out", "root.not.inputs.a", "root.not.g.inputs.a", "root.not.g.inputs.b"]
```
